function executeREST(operationUrl, method, body)
{
	System.debug("operation: " + operationUrl + ", Request Type: " + method);
	var request = restHost.createRequest(method, operationUrl, body);
	request.contentType = "application/json";
	request.setHeader("Accept", "application/json");
	
	var response = request.execute();
	System.debug("response: " + response);
	System.debug("statusCode: " + response.statusCode);
	if (response.statusCode == 200 || response.statusCode == 201)
	{
		var ABobj = System.getModule("com.dedicatedcloud.library.rest").getJSONFromString(response.contentAsString);
		var testjson = JSON.stringify(ABobj);
		System.log("TEST JSON LOG IS "+testjson);
		//System.debug("TEST JSON DEBUG LOG IS "+testjson);
		//throw "Forcefully failed here as part of testing by Akshay Bansal";
		if (response.contentAsString == null || response.contentAsString == "null" || response.contentAsString == "")
			return System.getModule("com.dedicatedcloud.library.rest").getJSONFromString('{}');
		else
			return System.getModule("com.dedicatedcloud.library.rest").getJSONFromString(response.contentAsString);
	}
	else
	{
		var json = System.getModule("com.dedicatedcloud.library.rest").getJSONFromString(response.contentAsString);
		System.error('REST request failed: ' + json.message);
		throw 'REST request failed: ' + json.message;
	}
}
